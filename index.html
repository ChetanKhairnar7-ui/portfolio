Transaction Handling

from django.db import transaction
from decimal import Decimal

# select_for_update locks the row
# preventing race conditions on stock
@transaction.atomic
def place_order(user, cart_items):
    order = Order.objects.create(
        user=user, status='PENDING'
    )
    total = Decimal('0.00')
    for item in cart_items:
        product = Product.objects\
            .select_for_update()\
            .get(id=item.product_id)
        if product.stock_count < item.qty:
            raise ValidationError(
                f"Out of stock: {product.title}"
            )
        product.stock_count -= item.qty
        product.save()
        OrderItem.objects.create(
            order=order,
            product=product,
            quantity=item.qty,
            unit_price=product.price
        )
        total += product.price * item.qty
    order.total_amount = total
    order.save()
    return order
